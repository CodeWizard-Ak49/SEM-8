#include <iostream>
#include <vector>
#include <omp.h>
#include <climits>

using namespace std;

int main() {

    vector<int> arr = {10, 20, 5, 40, 15, 35, 50, 25};

    int n = arr.size();

    int sum = 0;
    int min_val = INT_MAX;
    int max_val = INT_MIN;

    // Parallel Reduction
    #pragma omp parallel for reduction(+:sum) reduction(min:min_val) reduction(max:max_val)
    for (int i = 0; i < n; i++) {

        sum += arr[i];

        if (arr[i] < min_val)
            min_val = arr[i];

        if (arr[i] > max_val)
            max_val = arr[i];
    }

    double average = (double)sum / n;

    cout << "Array Elements:\n";

    for (int x : arr)
        cout << x << " ";

    cout << "\n\n";

    cout << "Minimum Value = " << min_val << endl;
    cout << "Maximum Value = " << max_val << endl;
    cout << "Sum = " << sum << endl;
    cout << "Average = " << average << endl;

    return 0;
}
