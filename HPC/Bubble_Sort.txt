#include <iostream>
#include <vector>
#include <omp.h>

using namespace std;

void parallelBubbleSort(vector<int>& arr) {

    int n = arr.size();

    for (int phase = 0; phase < n; phase++) {

        // Even phase
        if (phase % 2 == 0) {

            #pragma omp parallel for
            for (int i = 0; i < n - 1; i += 2) {

                if (arr[i] > arr[i + 1]) {
                    swap(arr[i], arr[i + 1]);
                }
            }
        }

        // Odd phase
        else {

            #pragma omp parallel for
            for (int i = 1; i < n - 1; i += 2) {

                if (arr[i] > arr[i + 1]) {
                    swap(arr[i], arr[i + 1]);
                }
            }
        }
    }
}

void printArray(vector<int>& arr) {

    for (int x : arr)
        cout << x << " ";

    cout << endl;
}

int main() {

    vector<int> arr = {9, 5, 1, 8, 3, 7, 2, 6, 4};

    cout << "Original Array:\n";
    printArray(arr);

    parallelBubbleSort(arr);

    cout << "Sorted Array:\n";
    printArray(arr);

    return 0;
}
